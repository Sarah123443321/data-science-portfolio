race_data = df_analysis[
    (df_analysis["Year"] == "2024") &
    (df_analysis["Demographics_Type"] == "RaceEthnicity")
].copy()

race_table = race_data.pivot_table(
    index="Demographics_Value",
    columns="Question",
    values="Percent",
    aggfunc="first"
).reset_index()

race_table.plot(
    x="Demographics_Value",
    y=[anxiety_freq, depression_freq, treatment],
    kind="bar",
    figsize=(12, 7)
)

plt.title("Self-Reported Mental Health Symptoms and Treatment by Race/Ethnicity, 2024")

plt.ylabel("Percent")

plt.legend(
    ["Anxiety Symptoms", "Depression Symptoms", "Treatment"],
    title="Measure"
)
